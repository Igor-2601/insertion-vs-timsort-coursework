cmake_minimum_required(VERSION 3.16)
project(insertion_vs_timsort LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(sort_benchmark src/benchmark.cpp)
add_executable(sort_tests tests/test_sorts.cpp)
foreach(target sort_benchmark sort_tests)
    target_include_directories(${target} PRIVATE include)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /O2)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -O2)
    endif()
endforeach()
enable_testing()
add_test(NAME correctness_and_stability COMMAND sort_tests)
